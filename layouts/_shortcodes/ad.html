Accept 0-100 (percent) or 0-1 (fraction). Applied as CSS max-width:NN% which preserves the aspect ratio and scales the image down relative to the column. */ -}} {{- $scalePercent := "" }} {{- if $scale }} {{- $n := float $scale }} {{- if gt $n 0.0 }} {{- if le $n 1.0 }} {{- $scalePercent = printf "%.0f%%" (mul $n 100.0) }} {{- else if le $n 100.0 }} {{- $scalePercent = printf "%.0f%%" $n }} {{- end }} {{- end }} {{- end }} {{- /* Build the width/height attributes. */ -}} {{- $attr := "" }} {{- if $widthAttr }}{{ $attr = printf "%s width=\"%s\"" $attr $widthAttr }}{{ end }} {{- if $heightAttr }}{{ $attr = printf "%s height=\"%s\"" $attr $heightAttr }}{{ end }} {{- /* CSS size rules: exact w/h caps, plus proportional scale. */ -}} {{- $css := "" }} {{- if $w }}{{ $css = printf "%s max-width:%s;" $css $w }}{{ end }} {{- if $h }}{{ $css = printf "%s max-height:%s;" $css $h }}{{ end }} {{- if $scalePercent }}{{ $css = printf "%s max-width:%s; height:auto;" $css $scalePercent }}{{ end }}
