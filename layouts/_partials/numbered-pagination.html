{{- /* Numbered pagination bar: ‹ Previous 1 … 4 … 9 Next › Context: any Hugo paginator (must expose TotalPages, PageNumber, Pagers, HasPrev/HasNext, Prev/Next). Shows the first page, the last page, and a window of pages around the current one; a single "…" marks each gap. The current page gets .active. Hidden entirely when there is only one page. */}} {{- if gt .TotalPages 1 }} {{- $cur := .PageNumber }} {{- $total := .TotalPages }}
{{- if .HasPrev }}‹ Previous{{ end }} {{- $prev := 0 }} {{- range .Pagers }} {{- $n := .PageNumber }} {{- $show := or (eq $n 1) (eq $n $total) (and (ge $n (sub $cur 2)) (le $n (add $cur 2))) }} {{- if $show }} {{- if and (gt $prev 0) (gt $n (add $prev 1)) }}…{{ end }} {{ $n }} {{- $prev = $n }} {{- end }} {{- end }} {{- if .HasNext }}Next ›{{ end }}
{{- end }}
